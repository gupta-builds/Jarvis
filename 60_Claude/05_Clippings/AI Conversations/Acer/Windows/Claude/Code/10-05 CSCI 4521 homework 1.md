### Helper functions

Same structure as lecture 1.3: a classifier is a function `classify(X_train, y_train, X_test)` that returns predictions for `X_test`.
KNN normalizes with the **training** mean and standard deviation only, so no information from the test data leaks into the model.

code("""
def train_test_split(X, y, test_percent):
  numRows = y.size                             # number of rows in the dataset
  splitPoint = int((1 - test_percent) * numRows)
  p = np.random.permutation(numRows)           # shuffle the rows
  X = X[p]
  y = y[p]

  X_train, y_train = X[:splitPoint], y[:splitPoint]  # training rows
  X_test, y_test = X[splitPoint:], y[splitPoint:]    # testing rows (no overlap with training)
  return X_train, X_test, y_train, y_test
""")
code("""
def knn_classifier(k):
  def knn_classify(x_train, y_train, x_test):
    # z-score normalization using the training data only
    mean = x_train.mean(axis=0)
    std = x_train.std(axis=0)

    knn = KNeighborsClassifier(n_neighbors=k)
    knn.fit((x_train - mean) / std, y_train)
    return knn.predict((x_test - mean) / std)
  return knn_classify

def all_phishing_classify(x_train, y_train, x_test):
  return np.ones(len(x_test), dtype=int)   # says every webpage is phishing

def no_phishing_classify(x_train, y_train, x_test):
  return np.zeros(len(x_test), dtype=int)  # says no webpage is phishing
""")
md("""
Metrics, with phishing (1) as the positive class. If a classifier never predicts phishing, precision is 0/0 (undefined); it is reported as 0.
""")
code("""
def confusion_counts(y_true, y_pred):
  tp = np.sum((y_pred == 1) & (y_true == 1))  # phishing caught
  fp = np.sum((y_pred == 1) & (y_true == 0))  # legitimate page flagged
  fn = np.sum((y_pred == 0) & (y_true == 1))  # phishing missed
  tn = np.sum((y_pred == 0) & (y_true == 0))  # legitimate page passed
  return tp, fp, fn, tn

def precision(y_true, y_pred):
  tp, fp, fn, tn = confusion_counts(y_true, y_pred)
  if tp + fp == 0:
    return 0.0
  return tp / (tp + fp)

def recall(y_true, y_pred):
  tp, fp, fn, tn = confusion_counts(y_true, y_pred)
  if tp + fn == 0:
    return 0.0
  return tp / (tp + fn)

def accuracy(y_true, y_pred):
  return np.mean(y_true == y_pred)

def f1_score(y_true, y_pred):
  p = precision(y_true, y_pred)
  r = recall(y_true, y_pred)
  if p + r == 0:
    return 0.0
  return 2 * p * r / (p + r)

def evaluate(classify, X_train, y_train, X_eval, y_eval):
  pred = classify(X_train, y_train, X_eval)
  return {"precision": precision(y_eval, pred), "recall": recall(y_eval, pred),
          "accuracy": accuracy(y_eval, pred), "f1": f1_score(y_eval, pred)}
""")

  train_total = {"precision": 0, "recall": 0, "accuracy": 0, "f1": 0}

  for _ in range(count):

def train_test_split(X, y, test_percent):
  numRows = y.size
  splitPoint = int((1 - test_percent) * numRows)
  p = np.random.permutation(numRows)   # shuffle the rows
  X = X[p]
  y = y[p]
  return X[:splitPoint], X[splitPoint:], y[:splitPoint], y[splitPoint:]

def knn_classifier(k):
  def knn_classify(x_train, y_train, x_test):
    mean = x_train.mean(axis=0)        # z-score normalization from training data only
    std = x_train.std(axis=0)
    knn = KNeighborsClassifier(n_neighbors=k)
    knn.fit((x_train - mean) / std, y_train)
    return knn.predict((x_test - mean) / std)
  return knn_classify

  pPr.append(bdr)

    p = doc.add_paragraph()
    p.alignment = WD_ALIGN_PARAGRAPH.CENTER
    p.paragraph_format.keep_with_next = True
    p.paragraph_format.space_after = Pt(0)

    c = doc.add_paragraph(l)

  p = doc.add_paragraph(l)
  if title_done < 3:
    p.alignment = WD_ALIGN_PARAGRAPH.CENTER
    p.paragraph_format.space_after = Pt(0)

    if title_done == 3:

doc.save(out)
