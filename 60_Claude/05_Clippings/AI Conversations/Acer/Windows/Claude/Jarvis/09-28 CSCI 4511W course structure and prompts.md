    $outFile = Join-Path $outDir ($f.BaseName -replace '[^\w\- ]','_')

    $slideEntries = $zip.Entries | Where-Object { $_.FullName -match '^ppt/slides/slide(\d+)\.xml$' } |
        Sort-Object { [int]([regex]::Match($_.FullName,'slide(\d+)\.xml').Groups[1].Value) }

    foreach ($entry in $slideEntries) {
        $num = [regex]::Match($entry.FullName,'slide(\d+)\.xml').Groups[1].Value
        $sb.AppendLine("--- Slide $num ---") | Out-Null
        $reader = New-Object System.IO.StreamReader($entry.Open())
        $xml = $reader.ReadToEnd()
        $reader.Close()
        # extract all <a:t>...</a:t> text runs in order
        $matches = [regex]::Matches($xml, '<a:t>(.*?)</a:t>')
        foreach ($m in $matches) {
            $text = [System.Net.WebUtility]::HtmlDecode($m.Groups[1].Value)
            if ($text.Trim().Length -gt 0) { $sb.AppendLine($text) | Out-Null }
        }
        # try matching notes slide
        $notesEntry = $zip.Entries | Where-Object { $_.FullName -eq "ppt/notesSlides/notesSlide$num.xml" }
        if ($notesEntry) {
            $nreader = New-Object System.IO.StreamReader($notesEntry.Open())
            $nxml = $nreader.ReadToEnd()
            $nreader.Close()
            $nmatches = [regex]::Matches($nxml, '<a:t>(.*?)</a:t>')
            $noteTexts = @()
            foreach ($nm in $nmatches) {
                $t = [System.Net.WebUtility]::HtmlDecode($nm.Groups[1].Value)
                if ($t.Trim().Length -gt 0) { $noteTexts += $t }
            }
            if ($noteTexts.Count -gt 0) {
                $sb.AppendLine("[Speaker notes: " + ($noteTexts -join " ") + "]") | Out-Null
            }
        }
    }
    $zip.Dispose()
    $sb.ToString() | Out-File -FilePath $outFile -Encoding utf8
    Write-Output "Wrote $outFile ($((Get-Item $outFile).Length) bytes)"
