import json, sys, glob, os, collections

def analyze(path):
    lines = 0
    by_id = collections.defaultdict(list)  # message.id -> list of (lineno, usage_json, uuid, stop_reason, content_types)
    no_id = 0
    for i, raw in enumerate(open(path, encoding='utf-8')):
        raw = raw.strip()
        if not raw:
            continue
        try:
            o = json.loads(raw)
        except Exception:
            continue
        if o.get('type') != 'assistant':
            continue
        m = o.get('message') or {}
        u = m.get('usage')
        if not u:
            continue
        lines += 1
        mid = m.get('id')
        if not mid:
            no_id += 1
            continue
        ctypes = [b.get('type') for b in (m.get('content') or []) if isinstance(b, dict)]
        by_id[mid].append((i + 1, json.dumps(u, sort_keys=True), o.get('uuid'), m.get('stop_reason'), ctypes, o.get('requestId')))
    return lines, no_id, by_id

def totals(usages):
    t = collections.Counter()
    for u in usages:
        u = json.loads(u)
        t['input'] += u.get('input_tokens') or 0
        t['output'] += u.get('output_tokens') or 0
        t['cache_read'] += u.get('cache_read_input_tokens') or 0
        t['cache_create'] += u.get('cache_creation_input_tokens') or 0
    return t

def main(paths, verbose):
    G_lines = 0; G_ids = 0
    naive = collections.Counter(); dedup_first = collections.Counter(); dedup_last = collections.Counter()
    multi_ids = 0; multi_identical = 0; multi_distinct = 0
    example = None
    for p in paths:
        lines, no_id, by_id = analyze(p)
        G_lines += lines; G_ids += len(by_id)
        for mid, ents in by_id.items():
            usages = [e[1] for e in ents]
            naive.update(totals(usages))
            dedup_first.update(totals([usages[0]]))
            dedup_last.update(totals([usages[-1]]))
            if len(ents) > 1:

                if len(set(usages)) == 1:
                    multi_identical += 1
                else:
